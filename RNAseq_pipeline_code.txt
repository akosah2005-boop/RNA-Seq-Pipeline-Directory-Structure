====================================================
Step 2: Quality Control Script
File: scripts/run_fastqc.sh
====================================================

#!/bin/bash

mkdir -p qc

for sample in raw_data/*.fastq.gz
do
    fastqc $sample -o qc/
done

multiqc qc/ -o qc/

====================================================
Step 3: Adapter and Quality Trimming
File: scripts/trim_reads.sh
====================================================

#!/bin/bash

mkdir -p trimmed

for sample in raw_data/*_R1.fastq.gz
do
    base=$(basename $sample "_R1.fastq.gz")

    trim_galore \
    --paired \
    raw_data/${base}_R1.fastq.gz \
    raw_data/${base}_R2.fastq.gz \
    -o trimmed/
done

====================================================
Step 4: Build STAR Genome Index
File: scripts/build_star_index.sh
====================================================

#!/bin/bash

mkdir -p reference/star_index

STAR \
--runThreadN 8 \
--runMode genomeGenerate \
--genomeDir reference/star_index \
--genomeFastaFiles reference/genome.fa \
--sjdbGTFfile reference/genes.gtf \
--sjdbOverhang 100

====================================================
Step 5: Alignment with STAR
File: scripts/align_reads.sh
====================================================

#!/bin/bash

mkdir -p aligned

for sample in trimmed/*_val_1.fq.gz
do
    base=$(basename $sample "_R1_val_1.fq.gz")

    STAR \
    --runThreadN 8 \
    --genomeDir reference/star_index \
    --readFilesIn \
    trimmed/${base}_R1_val_1.fq.gz \
    trimmed/${base}_R2_val_2.fq.gz \
    --readFilesCommand zcat \
    --outSAMtype BAM SortedByCoordinate \
    --outFileNamePrefix aligned/${base}_
done

====================================================
Step 7: Differential Expression in R
File: scripts/deseq2_analysis.R
====================================================

library(DESeq2)
library(ggplot2)
library(pheatmap)

# Load count matrix
counts <- read.delim(
    "counts/gene_counts.txt",
    comment.char="#",
    row.names=1
)

counts <- counts[,6:ncol(counts)]

# Metadata
sample_info <- data.frame(
    row.names = colnames(counts),
    condition = c("Control","Control","Treatment","Treatment")
)

# Create DESeq dataset
dds <- DESeqDataSetFromMatrix(
    countData = counts,
    colData = sample_info,
    design = ~ condition
)

# Filter low counts
dds <- dds[rowSums(counts(dds)) > 10,]

# Run differential expression
dds <- DESeq(dds)

res <- results(dds)

# Save results
write.csv(as.data.frame(res),
          file="results/deseq2_results.csv")

# Volcano Plot
res_df <- as.data.frame(res)

png("results/volcano_plot.png", width=1000, height=800)

plot(
    res_df$log2FoldChange,
    -log10(res_df$pvalue),
    pch=20,
    main="Volcano Plot",
    xlab="Log2 Fold Change",
    ylab="-Log10 p-value"
)

dev.off()

# PCA Plot
vsd <- vst(dds)

png("results/pca_plot.png", width=1000, height=800)

plotPCA(vsd, intgroup="condition")

dev.off()

Rscript scripts/deseq2_analysis.R
